cmake_minimum_required(VERSION 3.18...3.31)
project(permutohedral_segmentation VERSION 0.2.0 LANGUAGES CXX)

option(PERMUTOHEDRAL_BUILD_PYTHON "Build the nanobind extension" ON)
option(PERMUTOHEDRAL_BUILD_TESTS "Build standalone native tests" OFF)
option(PERMUTOHEDRAL_SANITIZERS "Enable ASan/UBSan for GNU/Clang native tests" OFF)

add_library(permutohedral_core STATIC cpp/src/lattice.cpp cpp/src/regions.cpp)
target_include_directories(permutohedral_core PUBLIC cpp/include)
target_compile_features(permutohedral_core PUBLIC cxx_std_17)
set_target_properties(permutohedral_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(permutohedral_core PRIVATE /W4 /fp:strict)
else()
  target_compile_options(permutohedral_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(PERMUTOHEDRAL_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "This sanitizer option supports GNU/Clang only")
  endif()
  target_compile_options(permutohedral_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(permutohedral_core PUBLIC -fsanitize=address,undefined)
endif()

if(PERMUTOHEDRAL_BUILD_PYTHON)
  find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(nanobind CONFIG REQUIRED)
  nanobind_add_module(_native cpp/src/bindings.cpp)
  target_link_libraries(_native PRIVATE permutohedral_core)
  install(TARGETS _native LIBRARY DESTINATION permutohedral_segmentation
                          RUNTIME DESTINATION permutohedral_segmentation)
endif()

if(PERMUTOHEDRAL_BUILD_TESTS)
  enable_testing()
  add_executable(test_core tests/native/test_core.cpp)
  target_link_libraries(test_core PRIVATE permutohedral_core)
  add_test(NAME core COMMAND test_core)
  # Explicit test-only FFI. Never installed in the Python package.
  add_library(permutohedral_test_bridge SHARED tests/native/bridge.cpp)
  target_link_libraries(permutohedral_test_bridge PRIVATE permutohedral_core)
  set_target_properties(permutohedral_test_bridge PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
